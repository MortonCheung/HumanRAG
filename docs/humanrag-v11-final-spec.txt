比如：
export function pickIntroSubgraph(
  preset: ConstellationPreset,
  graph: KnowledgeGraphData,
  seedId: string,
) {
  const selected: string[] = [];
  const queue = [seedId];
  const seen = new Set<string>();

  while (queue.length && selected.length < preset.points.length) {
    const id = queue.shift()!;

    if (seen.has(id)) continue;

    seen.add(id);
    selected.push(id);

    graph.edges
      .filter((e) => e.source === id || e.target === id)
      .forEach((e) => {
        const next = e.source === id ? e.target : e.source;

        if (!seen.has(next)) queue.push(next);
      });
  }

  const ids = new Set(selected);

  return {
    nodeIds: selected,
    edges: graph.edges.filter(
      (edge) => ids.has(edge.source) && ids.has(edge.target),
    ),
  };
}

步骤 11.2：自然语言评分
第一版不需要 LLM。
先做确定性、可解释、比赛稳定的：
function scorePoint(
  point: KnowledgePoint,
  query: GoalQuery,
) {
  let score = 0;

  const text = [
    point.name,
    point.description,
    ...point.tags,
    ...point.learningObjectives,
    ...point.recommendedContent,
  ].join(' ').toLowerCase();

  for (const term of query.goalTerms) {
    if (point.name.includes(term)) score += 8;
    else if (point.tags.some(t => t.includes(term))) score += 5;
    else if (text.includes(term)) score += 2;
  }

  for (const term of query.weakTerms) {
    if (text.includes(term)) score += 3.5;
  }

  for (const term of query.interestTerms) {
    if (text.includes(term)) score += 1.5;
  }

  for (const term of query.strongTerms) {
    if (text.includes(term)) score += 0.5;
  }

  return score;
}
解释：
- “薄弱”不是不选，反而应当增加相关权重。
- “擅长”不删除，只降低学习优先级。
- 兴趣可以影响边缘节点。

function propagateScores(
  scores: Map<string, number>,
  relations: KnowledgeRelation[],
) {
  const next = new Map(scores);

  for (const relation of relations) {
    const source = scores.get(relation.sourcePointId) ?? 0;
    const target = scores.get(relation.targetPointId) ?? 0;

    if (relation.type === 'prerequisite') {
      next.set(
        relation.sourcePointId,
        Math.max(
          next.get(relation.sourcePointId) ?? 0,
          target * 0.72,
        ),
      );
    }

    if (relation.type === 'hierarchy') {
      next.set(
        relation.targetPointId,
        Math.max(
          next.get(relation.targetPointId) ?? 0,
          source * 0.66,
        ),
      );
    }

    if (relation.type === 'related') {
      next.set(
        relation.targetPointId,
        Math.max(
          next.get(relation.targetPointId) ?? 0,
          source * 0.30,
        ),
      );
    }
  }

  return next;
}

export function createTree(
  libraryId: string,
  input: CreateTreeInput,
): KnowledgeTree {
  const state = migrateV9();

  if (state.library.id !== libraryId) {
    throw new Error('未找到这个知识库。');
  }

  const existingTrees = [
    ...state.trees,
    ...state.userTrees,
  ];

  if (
    checkDuplicateTreeName(
      input.identity.name,
      existingTrees,
    )
  ) {
    throw new Error(
      '这个知识库中已经有同名知识树。',
    );
  }

  const validPointIds = [
    ...new Set(input.pointIds ?? []),
  ];

  const pointSet = new Set(
    state.points.map((p) => p.id),
  );

  if (
    validPointIds.some(id => !pointSet.has(id))
  ) {
    throw new Error(
      '知识树包含不存在的知识点。',
    );
  }

  const now = new Date().toISOString();

  const tree: KnowledgeTree = {
    id: `tree-${crypto.randomUUID()}`,
    libraryId,
    name: input.identity.name.trim(),
    description: input.identity.description,
    color: input.identity.color,
    ownerType: 'user',
    pointIds: validPointIds,
    createdAt: now,
    updatedAt: now,
  };

  const memberships = [
    ...state.memberships,
    ...deriveTreeMemberships(
      tree.id,
      validPointIds,
      state.relations,
    ),
  ];

  const nextLibrary = {
    ...state.library,
    treeIds: [
      ...state.library.treeIds,
      tree.id,
    ],
  };

  const nextUserTrees = [
    ...state.userTrees,
    tree,
  ];

  const result = saveV9State(
    nextLibrary,
    state.trees,
    state.points,
    state.relations,
    memberships,
    nextUserTrees,
  );

  if (!result.ok) {
    throw new Error(result.error);
  }

  return tree;
}

  treeId: string,
): PracticePlan | null {
  const tree = getTree(treeId);

  if (!tree) return null;

  const questionIds = dedupe(
    tree.pointIds.flatMap((pointId) =>
      contentRepository
        .getQuestionsForNode(pointId)
        .map((q) => q.id),
    ),
  ).slice(0, 28);

  if (!questionIds.length) {
    return null;

  return {
    id: `tree:${tree.id}`,
    mode: 'goal',
    title: `${tree.name} · 能力验证`,
    description: '',
    sourceLabel: tree.name,
    questionIds,
    estimatedMinutes:

  };

export type LearningState =
  | 'unknown'
  | 'learning'
  | 'needs-reinforcement'
  | 'needs-verification'
  | 'verified';

export function deriveLearningState(
  pointId: string,
  learnerId: string,
): LearningState {
  const progress =
    useProgressStore.getState();

  const status =
    learningStatusFromEvidence(
      progress.evidenceRecords,
      pointId,
      learnerId,
    );

  if (
    status.status === 'passed'
  ) {
    return 'verified';
  }

  if (
    status.status === 'needs-work'
  ) {
    return 'needs-reinforcement';
  }

  if (
    status.status === 'assisted'
  ) {
    return 'needs-verification';
  }

  return 'unknown';
}
